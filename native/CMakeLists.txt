cmake_minimum_required(VERSION 3.21)
project(MirivueKIO LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Xml)
add_library(mirivue-item-presentation SHARED item-presentation.cpp natural-sort.cpp)
target_link_libraries(mirivue-item-presentation PRIVATE Qt6::Widgets)
install(TARGETS mirivue-item-presentation LIBRARY DESTINATION lib/mirivue-kde/tools)
option(MIRIVUE_PRESENTATION_ONLY "Build the Qt presentation tests without KF6" OFF)
if(MIRIVUE_PRESENTATION_ONLY)
    return()
endif()
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
find_package(KF6GuiAddons REQUIRED)
find_package(KF6WidgetsAddons REQUIRED)
find_package(KF6TextWidgets REQUIRED)
find_package(KF6I18n REQUIRED)
find_package(KF6KIO REQUIRED)
find_package(KF6Config REQUIRED)
find_package(KF6ColorScheme REQUIRED)
find_package(KF6CoreAddons REQUIRED)
find_package(KF6IconThemes REQUIRED)
find_package(KF6WindowSystem REQUIRED)

add_subdirectory(dolphin-engine)
install(TARGETS mirivue-dolphin-view LIBRARY DESTINATION lib/mirivue-kde/tools)
option(MIRIVUE_VIEWS_ONLY "Build native view engines for automated GUI tests" OFF)
option(MIRIVUE_NAVIGATOR_TESTS "Build the native breadcrumb rendering checks" OFF)
add_executable(mirivue-kio-list kio-list.cpp)
target_link_libraries(mirivue-kio-list PRIVATE Qt6::Core KF6::KIOCore)
if(MIRIVUE_NAVIGATOR_TESTS)
    add_library(mirivue-open-with-reference SHARED tests/open-with-reference.cpp)
    target_link_libraries(mirivue-open-with-reference PRIVATE Qt6::Widgets KF6::KIOWidgets KF6::KIOCore)
    add_executable(mirivue-view-palette-check tests/view-palette-check.cpp)
    target_link_libraries(mirivue-view-palette-check PRIVATE mirivue-dolphin-view Qt6::Widgets KF6::ColorScheme)
    add_executable(mirivue-navigator-check tests/navigator-check.cpp path-navigator.cpp)
    target_include_directories(mirivue-navigator-check PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(mirivue-navigator-check PRIVATE Qt6::Widgets KF6::KIOFileWidgets)
endif()
if(MIRIVUE_VIEWS_ONLY)
    return()
endif()

find_package(KF6JobWidgets REQUIRED)
find_package(KF6CoreAddons REQUIRED)
find_package(KF6IconThemes REQUIRED)
find_package(KF6WindowSystem REQUIRED)
find_package(KF6XmlGui REQUIRED)
find_package(KF6ConfigWidgets REQUIRED)
find_package(KF6Service REQUIRED)
find_package(KF6Parts REQUIRED)
find_package(KF6NewStuff REQUIRED)

add_executable(mirivue-kio-open kio-open.cpp)
target_link_libraries(mirivue-kio-open PRIVATE Qt6::Widgets KF6::KIOGui KF6::KIOWidgets)

add_executable(mirivue-kio-applications kio-applications.cpp)
target_link_libraries(mirivue-kio-applications PRIVATE Qt6::Widgets KF6::KIOGui KF6::KIOWidgets KF6::Service)

add_executable(mirivue-kio-empty-trash kio-empty-trash.cpp)
target_link_libraries(mirivue-kio-empty-trash PRIVATE Qt6::Widgets KF6::KIOCore)

add_executable(mirivue-kio-transfer kio-transfer.cpp)
target_link_libraries(mirivue-kio-transfer PRIVATE Qt6::Widgets KF6::KIOWidgets)

add_executable(mirivue-job-progress job-progress.cpp)
target_link_libraries(mirivue-job-progress PRIVATE Qt6::Core Qt6::Gui KF6::CoreAddons KF6::JobWidgets)

add_executable(mirivue-kio-service kio-service.cpp)
target_link_libraries(mirivue-kio-service PRIVATE Qt6::Widgets KF6::KIOWidgets)

add_executable(mirivue-kio-preview kio-preview.cpp)
target_link_libraries(mirivue-kio-preview PRIVATE Qt6::Core Qt6::Gui KF6::KIOGui)

add_executable(mirivue-kio-places kio-places.cpp)
target_link_libraries(mirivue-kio-places PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets KF6::KIOFileWidgets)

add_library(mirivue-kio-places-widget SHARED kio-places-widget.cpp)
target_sources(mirivue-kio-places-widget PRIVATE terminal-widget.cpp path-navigator.cpp context-menu-settings.cpp toolbar-editor.cpp desktop-dialogs.cpp open-with-menu.cpp)
target_link_libraries(mirivue-kio-places-widget PRIVATE Qt6::Widgets Qt6::Xml KF6::KIOFileWidgets KF6::KIOWidgets KF6::JobWidgets KF6::ConfigCore KF6::ConfigWidgets KF6::ColorScheme KF6::IconThemes KF6::XmlGui KF6::Parts KF6::CoreAddons KF6::WindowSystem KF6::Service KF6::NewStuffWidgets)

install(TARGETS mirivue-kio-list mirivue-kio-open mirivue-kio-applications mirivue-kio-empty-trash mirivue-kio-transfer mirivue-kio-service mirivue-kio-preview mirivue-kio-places mirivue-job-progress
        RUNTIME DESTINATION bin)

# Disposable providers for integration tests. Never installed by package().
if(MIRIVUE_NAVIGATOR_TESTS)
    foreach(provider compressfileitemaction extractfileitemaction movetonewfolderitemaction)
        add_library(mirivue-test-${provider} MODULE tests/context-action-plugin.cpp)
        target_link_libraries(mirivue-test-${provider} PRIVATE Qt6::Widgets KF6::KIOWidgets KF6::CoreAddons)
        set_target_properties(mirivue-test-${provider} PROPERTIES PREFIX "" OUTPUT_NAME ${provider}
            LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-plugins/kf6/kfileitemaction")
    endforeach()
endif()
